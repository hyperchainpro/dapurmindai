   109→function mapToCategory(category: string): string {
   110→  const lc = category.toLowerCase();
   111→  if (lc.includes('sayur') || lc.includes('kol') || lc.includes('tauge') || lc.includes('kangkung') || lc.includes('sawi') || lc.includes('daun bawang') || lc.includes('seledri') || lc.includes('nangka') || lc.includes('kacang panjang') || lc.includes('labu') || lc.includes('melinjo') || lc.includes('kecipir')) return 'Sayuran';
   112→  if (lc.includes('protein') || lc.includes('ayam') || lc.includes('daging') || lc.includes('sapi') || lc.includes('telur') || lc.includes('tempe') || lc.includes('tahu') || lc.includes('ikan')) return 'Daging';
   113→  if (lc.includes('bumbu') || lc.includes('cabai') || lc.includes('bawang') || lc.includes('kunyit') || lc.includes('jahe') || lc.includes('lengkuas') || lc.includes('serai') || lc.includes('garam') || lc.includes('gula') || lc.includes('kecap') || lc.includes('santan') || lc.includes('merica') || lc.includes('sambal')) return 'Bumbu';
   114→  if (lc.includes('bahan utama') || lc.includes('nasi') || lc.includes('beras') || lc.includes('mie') || lc.includes('tepung') || lc.includes('air') || lc.includes('minyak') || lc.includes('susu')) return 'Bahan Pokok';
   115→  if (lc.includes('susu') || lc.includes('telur') || lc.includes('kelapa')) return 'Susu & Telur';
   116→  return 'Lainnya';
   117→}

   309→  const isAILoading = useAppStore((s) => s.isAILoading);
   310→
   311→  /* ── Local state ───────────────────────────────────── */
   312→  const [searchQuery, setSearchQuery] = useState('');
   313→  const [activeCategory, setActiveCategory] = useState('semua');
   314→  const [aiResults, setAiResults] = useState<AILinkResult[]>([]);
   315→  const [aiError, setAiError] = useState<string | null>(null);
   316→  const [aiProductName, setAiProductName] = useState('');
   317→  const [showAIDialog, setShowAIDialog] = useState(false);
   318→  const [isAILinkLoading, setIsAILinkLoading] = useState(false);
   319→
   320→  /* ── Derived state ─────────────────────────────────── */
   321→  const uncheckedItems = useMemo(
   322→    () => shoppingItems.filter((i) => !i.checked),
   323→    [shoppingItems]
   324→  );
   325→
   326→  const filteredItems = useMemo(() => {
   327→    let items = uncheckedItems;
   328→    if (activeCategory !== 'semua') {
   329→      items = items.filter((i) => mapToCategory(i.category) === activeCategory);
   330→    }
   331→    if (searchQuery.trim()) {
   332→      const q = searchQuery.toLowerCase();
   333→      items = items.filter((i) => i.name.toLowerCase().includes(q));
   334→    }
   335→    return items;
   336→  }, [uncheckedItems, activeCategory, searchQuery]);
   337→
   338→  // Group filtered items by category
   339→  const groupedItems = useMemo(() => {
   340→    const groups: Record<string, ShoppingItem[]> = {};
   341→    filteredItems.forEach((item) => {
   342→      const cat = mapToCategory(item.category);
   343→      if (!groups[cat]) groups[cat] = [];
   344→      groups[cat].push(item);
   345→    });
   346→    const order = ['Sayuran', 'Daging', 'Bumbu', 'Bahan Pokok', 'Susu & Telur', 'Lainnya'];
   347→    const sorted: [string, ShoppingItem[]][] = [];
   348→    order.forEach((cat) => {
   349→      if (groups[cat]) sorted.push([cat, groups[cat]]);
   350→    });
   351→    Object.keys(groups).forEach((cat) => {
   352→      if (!order.includes(cat)) sorted.push([cat, groups[cat]]);
   353→    });
   354→    return sorted;
   355→  }, [filteredItems]);
   356→
   357→  // Estimated total
   358→  const estimatedTotal = useMemo(
   359→    () => filteredItems.reduce((sum, i) => sum + (i.estimatedPrice ?? 15000), 0),
   360→    [filteredItems]
   361→  );
   362→
   363→  // Top recommended marketplace
   364→  const topMarketplace = useMemo(
   365→    () => AFFILIATE_MARKETPLACES.find((m) => m.rating === 5) ?? AFFILIATE_MARKETPLACES[0],
   366→    []
   367→  );
   368→
   369→  /* ── Handlers ──────────────────────────────────────── */
   370→  const handleBuyItem = useCallback((item: ShoppingItem) => {
   371→    const url = buildAffiliateUrl('tokopedia-now', item.name);
   372→    window.open(url, '_blank', 'noopener,noreferrer');
   373→  }, []);
   374→
   375→  const handleBuyAllTokopedia = useCallback(() => {
   376→    const names = filteredItems.map((i) => i.name);
   377→    if (names.length === 0) return;
   378→    const url = buildBulkAffiliateUrl('tokopedia-now', names);
   379→    window.open(url, '_blank', 'noopener,noreferrer');
   380→  }, [filteredItems]);
   381→
   382→  const handleBuyAllShopee = useCallback(() => {
   383→    const names = filteredItems.map((i) => i.name);
   384→    if (names.length === 0) return;
   385→    const url = buildBulkAffiliateUrl('shopee-segar', names);
   386→    window.open(url, '_blank', 'noopener,noreferrer');
   387→  }, [filteredItems]);
   388→
   389→  const handleBuyAllMulti = useCallback(() => {
   390→    const names = filteredItems.map((i) => i.name);
   391→    if (names.length === 0) return;
   392→    ['tokopedia-now', 'shopee-segar', 'sayurbox'].forEach((mpId, idx) => {
   393→      setTimeout(() => {
   394→        const url = buildBulkAffiliateUrl(mpId, names);
   395→        window.open(url, '_blank', 'noopener,noreferrer');
   396→      }, idx * 300);
   397→    });
   398→  }, [filteredItems]);
   399→
   400→  const handleSearchAllMarketplaces = useCallback(() => {
   401→    const query = searchQuery.trim() || 'bahan masakan';
   402→    AFFILIATE_MARKETPLACES.forEach((mp, idx) => {
   403→      setTimeout(() => {
   404→        const url = buildAffiliateUrl(mp.id, query);
   405→        window.open(url, '_blank', 'noopener,noreferrer');
   406→      }, idx * 500);
   407→    });
   408→  }, [searchQuery]);
   409→
   410→  const handleAIGenerateLinks = useCallback(async (productName: string) => {
   411→    setIsAILinkLoading(true);
   412→    setAiError(null);
   413→    setAiResults([]);
   414→    setAiProductName(productName);
   415→
   416→    try {
   417→      const response = await fetch('/api/affiliate/generate-link', {
   418→        method: 'POST',
   419→        headers: { 'Content-Type': 'application/json' },
   420→        body: JSON.stringify({
   421→          productName,
   422→          category: activeCategory !== 'semua' ? activeCategory : 'Lainnya',
   423→          context: 'marketplace-hub',
   424→        }),
   425→      });
   426→
   427→      const data = await response.json();
   428→
   429→      if (!response.ok) {
   430→        setAiError(data.error || 'Gagal generate tautan. Coba lagi.');
   431→        return;
   432→      }
   433→
   434→      if (data.links && data.links.length > 0) {
   435→        setAiResults(data.links);
   436→      } else {
   437→        setAiError('Tidak ada akun afiliasi aktif. Hubungi admin.');
   438→      }
   439→    } catch {
   440→      setAiError('Terjadi kesalahan jaringan. Periksa koneksi dan coba lagi.');

   442→      setIsAILinkLoading(false);
   443→    }

   449→    }
   450→  }, [aiProductName, handleAIGenerateLinks]);

   456→        <div className="flex flex-col pb-28">
   457→          {/* Header */}
   458→          <header className="sticky top-0 z-20 glass">
   459→            <div className="flex items-center gap-3 px-4 py-3">
   460→              <motion.button
   461→                whileTap={{ scale: 0.9 }}
   462→                onClick={goBack}
   463→                className="flex h-9 w-9 items-center justify-center rounded-full nm-raised transition-colors hover:bg-accent"
   464→                aria-label="Kembali"
   465→              >
   466→                <ArrowLeft className="h-4 w-4" />

   468→              <div className="flex items-center gap-2">
