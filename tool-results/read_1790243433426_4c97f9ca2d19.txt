   185→
   186→function getRelativeTime(dateStr?: string): string {
   187→  if (!dateStr) return 'Belum pernah';
   188→  const now = Date.now();
   189→  const then = new Date(dateStr).getTime();
   190→  const diffMs = now - then;
   191→  const diffSec = Math.floor(diffMs / 1000);
   192→  const diffMin = Math.floor(diffSec / 60);
   193→  const diffHour = Math.floor(diffMin / 60);
   194→  const diffDay = Math.floor(diffHour / 24);
   195→
   196→  if (diffSec < 60) return 'Baru saja';
   197→  if (diffMin < 60) return `${diffMin} menit lalu`;
   198→  if (diffHour < 24) return `${diffHour} jam lalu`;
   199→  if (diffDay < 30) return `${diffDay} hari lalu`;
   200→  return new Date(dateStr).toLocaleDateString('id-ID', { day: 'numeric', month: 'short', year: 'numeric' });
   201→}
   202→
   203→function getSuccessRate(agent: AiAgent): number {
   204→  if (agent.totalRequests === 0) return 100;
   205→  return Math.round(((agent.totalRequests - agent.failedRequests) / agent.totalRequests) * 100);
   206→}
   207→
   208→function getTokenPercentage(used: number, max: number): number {
   209→  if (max === 0) return 0;
   210→  return Math.min(100, Math.round((used / max) * 100));
   211→}
   212→
   213→function getTokenBarColor(pct: number): string {
   214→  if (pct >= 90) return 'bg-red-500';
   215→  if (pct >= 70) return 'bg-amber-500';
   216→  if (pct >= 40) return 'bg-blue-500';
   217→  return 'bg-emerald-500';
   218→}
   219→
   220→function formatTokens(n: number): string {
   221→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(1)}M`;
   222→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
   223→  return String(n);
   224→}
