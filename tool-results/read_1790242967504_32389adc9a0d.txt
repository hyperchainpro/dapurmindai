    61→/* ── Main Component ──────────────────────────────────────────────── */
    62→
    63→export function FinancialPlannerPage() {
    64→  const { t } = useTranslation();
    65→  const authUser = useAppStore((s) => s.authUser);
    66→  const goBack = useAppStore((s) => s.goBack);
    67→  const records = useAppStore((s) => s.financeRecords);
    68→  const budgets = useAppStore((s) => s.financeBudgets);
    69→  const goals = useAppStore((s) => s.financeGoals);
    70→  const loading = useAppStore((s) => s.isFinanceLoading);
    71→  const setRecords = useAppStore((s) => s.setFinanceRecords);
    72→  const setBudgets = useAppStore((s) => s.setFinanceBudgets);
    73→  const setGoals = useAppStore((s) => s.setFinanceGoals);
    74→  const addRecord = useAppStore((s) => s.addFinanceRecord);
    75→  const updateRecord = useAppStore((s) => s.updateFinanceRecord);
    76→  const removeRecord = useAppStore((s) => s.removeFinanceRecord);
    77→  const addBudget = useAppStore((s) => s.addFinanceBudget);
    78→  const updateBudget = useAppStore((s) => s.updateFinanceBudget);
    79→  const removeBudget = useAppStore((s) => s.removeFinanceBudget);
    80→  const addGoal = useAppStore((s) => s.addFinanceGoal);
    81→  const updateGoal = useAppStore((s) => s.updateFinanceGoal);
    82→  const removeGoal = useAppStore((s) => s.removeFinanceGoal);
    83→  const setLoading = useAppStore((s) => s.setFinanceLoading);
    84→  const userId = authUser?.id;
    85→
    86→  const [activeTab, setActiveTab] = useState<'records' | 'budgets' | 'goals' | 'report' | 'recurring'>('records');
    87→
    88→  // AI Advisor state
    89→  const [showAiDialog, setShowAiDialog] = useState(false);
    90→  const [aiQuestion, setAiQuestion] = useState('');
    91→  const [aiResponse, setAiResponse] = useState('');
    92→  const [aiLoading, setAiLoading] = useState(false);
    93→  const [aiHistory, setAiHistory] = useState<{ role: 'user' | 'ai'; content: string }[]>([]);
    94→
    95→  // Dialog open states
    96→  const [dlgRecord, setDlgRecord] = useState(false);
    97→  const [dlgBudget, setDlgBudget] = useState(false);
    98→  const [dlgGoal, setDlgGoal] = useState(false);
    99→  const [dlgSavings, setDlgSavings] = useState<FinanceGoal | null>(null);
   100→  const [dlgDelete, setDlgDelete] = useState<{ type: string; id: string } | null>(null);
   101→
   102→  // Record form
   103→  const [recType, setRecType] = useState<'income' | 'expense'>('expense');
   104→  const [recAmt, setRecAmt] = useState('');
   105→  const [recCat, setRecCat] = useState('');
   106→  const [recDesc, setRecDesc] = useState('');
   107→  const [recDate, setRecDate] = useState(() => new Date().toISOString().split('T')[0]);
   108→  const [editRecId, setEditRecId] = useState<string | null>(null);
   109→
   110→  // Budget form
   111→  const [budCat, setBudCat] = useState('');
   112→  const [budLimit, setBudLimit] = useState('');
   113→  const [budPeriod, setBudPeriod] = useState<'weekly' | 'monthly' | 'yearly'>('monthly');
   114→  const [editBudId, setEditBudId] = useState<string | null>(null);
   115→
   116→  // Goal form
   117→  const [goTitle, setGoTitle] = useState('');
   118→  const [goTarget, setGoTarget] = useState('');
   119→  const [goSaved, setGoSaved] = useState('');
   120→  const [goDeadline, setGoDeadline] = useState('');
   121→  const [goIcon, setGoIcon] = useState('🏠');
   122→  const [editGoId, setEditGoId] = useState<string | null>(null);
   123→
   124→  // Savings input
   125→  const [savAmt, setSavAmt] = useState('');
   126→
   127→  /* ── Fetch data ──────────────────────────────────── */
   128→  useEffect(() => {
   129→    if (!userId) return;
   130→    (async () => {
   131→      setLoading(true);
   132→      try {
   133→        const [r, b, g] = await Promise.all([
   134→          fetch(`/api/finance/records?userId=${userId}`),
   135→          fetch(`/api/finance/budgets?userId=${userId}`),
   136→          fetch(`/api/finance/goals?userId=${userId}`),
   137→        ]);
   138→        if (r.ok) { const json = await r.json(); setRecords(json.data ?? json); }
   139→        if (b.ok) { const json = await b.json(); setBudgets(json.data ?? json); }
   140→        if (g.ok) { const json = await g.json(); setGoals(json.data ?? json); }
   141→      } catch (e) { console.error(e); }
   142→      finally { setLoading(false); }
   143→    })();
   144→  }, [userId, setRecords, setBudgets, setGoals, setLoading]);
   145→
   146→  /* ── Monthly stats ───────────────────────────────── */
   147→  const now = new Date();
   148→  const mon = now.getMonth(), yr = now.getFullYear();
   149→  const monthlyIn = useMemo(() => records.filter((r) => { const d = new Date(r.date); return r.type === 'income' && d.getMonth() === mon && d.getFullYear() === yr; }).reduce((s, r) => s + r.amount, 0), [records, mon, yr]);
   150→  const monthlyOut = useMemo(() => records.filter((r) => { const d = new Date(r.date); return r.type === 'expense' && d.getMonth() === mon && d.getFullYear() === yr; }).reduce((s, r) => s + r.amount, 0), [records, mon, yr]);
   151→  const balance = monthlyIn - monthlyOut;
   152→
   153→  /* ── Grouped records ──────────────────────────────── */
   154→  const grouped = useMemo(() => {
   155→    const m = new Map<string, FinanceRecord[]>();
   156→    records.slice().sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()).forEach((r) => {
   157→      const k = r.date; if (!m.has(k)) m.set(k, []); m.get(k)!.push(r);
   158→    });
   159→    return Array.from(m.entries());
   160→  }, [records]);
   161→
   162→  /* ── Budgets with computed spent ─────────────────── */
   163→  const budgetsWithSpent = useMemo(() => budgets.map((b) => ({
   164→    ...b,
   165→    spent: records.filter((r) => r.type === 'expense' && r.category === b.category).reduce((s, r) => s + r.amount, 0),
   166→  })), [budgets, records]);
   167→
   168→  /* ── Form helpers ────────────────────────────────── */
   169→  const resetRec = () => { setEditRecId(null); setRecType('expense'); setRecAmt(''); setRecCat(''); setRecDesc(''); setRecDate(new Date().toISOString().split('T')[0]); };
   170→  const resetBud = () => { setEditBudId(null); setBudCat(''); setBudLimit(''); setBudPeriod('monthly'); };
   171→  const resetGo = () => { setEditGoId(null); setGoTitle(''); setGoTarget(''); setGoSaved(''); setGoDeadline(''); setGoIcon('🏠'); };
   172→
   173→  const openEditRec = (r: FinanceRecord) => { setEditRecId(r.id); setRecType(r.type); setRecAmt(String(r.amount)); setRecCat(r.category); setRecDesc(r.description); setRecDate(r.date); setDlgRecord(true); };
   174→  const openEditBud = (b: FinanceBudget) => { setEditBudId(b.id); setBudCat(b.category); setBudLimit(String(b.limitAmount)); setBudPeriod(b.period); setDlgBudget(true); };
   175→  const openEditGo = (g: FinanceGoal) => { setEditGoId(g.id); setGoTitle(g.title); setGoTarget(String(g.targetAmount)); setGoSaved(String(g.savedAmount)); setGoDeadline(g.deadline); setGoIcon(g.icon); setDlgGoal(true); };
   176→
   177→  /* ── Save handlers ───────────────────────────────── */
   178→  const saveRec = async () => {
   179→    if (!userId || !recAmt || !recCat) return;
   180→    try {
   181→      const body = { userId, type: recType, category: recCat, amount: Number(recAmt), description: recDesc, date: recDate };
   182→      const res = await fetch('/api/finance/records', { method: editRecId ? 'PUT' : 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(editRecId ? { id: editRecId, ...body } : body) });
   183→      if (res.ok) { const json = await res.json(); const record = json.data ?? json; editRecId ? updateRecord(editRecId, record) : addRecord(record); }
   184→      setDlgRecord(false); resetRec();
   185→    } catch (e) { console.error(e); }
   186→  };
   187→
   188→  const saveBud = async () => {
   189→    if (!userId || !budCat || !budLimit) return;
   190→    try {
   191→      const body = { userId, category: budCat, limitAmount: Number(budLimit), period: budPeriod };
   192→      const res = await fetch('/api/finance/budgets', { method: editBudId ? 'PUT' : 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(editBudId ? { id: editBudId, ...body } : body) });
   193→      if (res.ok) { const json = await res.json(); const budget = json.data ?? json; editBudId ? updateBudget(editBudId, budget) : addBudget(budget); }
   194→      setDlgBudget(false); resetBud();
   195→    } catch (e) { console.error(e); }
   196→  };
   197→
   198→  const saveGoal = async () => {
   199→    if (!userId || !goTitle || !goTarget) return;
   200→    try {
   201→      const body = { userId, title: goTitle, targetAmount: Number(goTarget), savedAmount: Number(goSaved) || 0, deadline: goDeadline, icon: goIcon };
   202→      const res = await fetch('/api/finance/goals', { method: editGoId ? 'PUT' : 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(editGoId ? { id: editGoId, ...body } : body) });
   203→      if (res.ok) { const json = await res.json(); const goal = json.data ?? json; editGoId ? updateGoal(editGoId, goal) : addGoal(goal); }
   204→      setDlgGoal(false); resetGo();
   205→    } catch (e) { console.error(e); }
   206→  };
   207→
   208→  const addSavings = async () => {
   209→    if (!dlgSavings || !userId || !savAmt) return;
   210→    const newSaved = dlgSavings.savedAmount + Number(savAmt);
   211→    try {
   212→      const res = await fetch('/api/finance/goals', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: dlgSavings.id, userId, savedAmount: newSaved }) });
   213→      if (res.ok) updateGoal(dlgSavings.id, { savedAmount: newSaved });
   214→      setDlgSavings(null); setSavAmt('');
   215→    } catch (e) { console.error(e); }
   216→  };
   217→
   218→  const handleDelete = async () => {
   219→    if (!dlgDelete || !userId) return;
   220→    const ep = dlgDelete.type === 'record' ? 'records' : dlgDelete.type === 'budget' ? 'budgets' : 'goals';
   221→    try {
   222→      const res = await fetch(`/api/finance/${ep}`, {
   223→        method: 'DELETE',
   224→        headers: { 'Content-Type': 'application/json' },
   225→        body: JSON.stringify({ id: dlgDelete.id, userId }),
   226→      });
   227→      if (res.ok) {
   228→        if (dlgDelete.type === 'record') removeRecord(dlgDelete.id);
   229→        else if (dlgDelete.type === 'budget') removeBudget(dlgDelete.id);
   230→        else removeGoal(dlgDelete.id);
   231→      }
   232→    } catch (e) { console.error(e); }
   233→    setDlgDelete(null);
   234→  };
   235→
   236→  /* ── AI Advisor ─────────────────────────────────── */
   237→  const handleAiSend = async () => {
   238→    if (!aiQuestion.trim() || !userId || aiLoading) return;
   239→    const q = aiQuestion.trim();
   240→    setAiQuestion('');
   241→    setAiHistory((prev) => [...prev, { role: 'user', content: q }]);
   242→    setAiLoading(true);
   243→    try {
   244→      const res = await fetch('/api/finance/ai-advice', {
   245→        method: 'POST',
   246→        headers: { 'Content-Type': 'application/json' },
   247→        body: JSON.stringify({ userId, question: q }),
   248→      });
   249→      if (res.ok) {
   250→        const json = await res.json();
   251→        const answer = json.data?.response ?? json.response ?? json.data?.answer ?? 'Maaf, saya tidak bisa menjawab saat ini.';
   252→        setAiHistory((prev) => [...prev, { role: 'ai', content: answer }]);
   253→      } else {
   254→        setAiHistory((prev) => [...prev, { role: 'ai', content: 'Maaf, terjadi kesalahan. Coba lagi nanti.' }]);
   255→      }
   256→    } catch {
   257→      setAiHistory((prev) => [...prev, { role: 'ai', content: 'Maaf, terjadi kesalahan koneksi. Coba lagi nanti.' }]);
   258→    } finally {
   259→      setAiLoading(false);
   260→    }
   261→  };
   262→
   263→  /* ── Render ───────────────────────────────────────── */
   264→  return (
   265→    <div className="min-h-screen bg-gradient-to-b from-emerald-50 via-white to-emerald-50/30 dark:from-emerald-950/40 dark:via-background dark:to-emerald-950/20">
   266→      <header className="sticky top-0 z-20 glass">
   267→        <div className="flex items-center gap-3 px-4 py-3">
   268→          <motion.button whileTap={{ scale: 0.9 }} onClick={goBack} className="flex h-9 w-9 items-center justify-center rounded-full nm-raised hover:bg-accent">
   269→            <ArrowLeft className="h-4 w-4" />
   270→          </motion.button>
   271→          <div className="flex items-center gap-2"><Wallet className="h-5 w-5 text-emerald-500" /><h1 className="text-lg font-bold tracking-tight">{t('finance.title')}</h1></div>
   272→        </div>
   273→      </header>
   274→
   275→      <div className="flex flex-col pb-24">
   276→        {/* ── Overview ──────────────────────────────── */}
   277→        {loading ? (
   278→          <div className="mx-4 mt-4 space-y-3"><Skeleton className="h-24 w-full rounded-xl" /><div className="flex gap-3"><Skeleton className="h-16 flex-1 rounded-xl" /><Skeleton className="h-16 flex-1 rounded-xl" /></div></div>
   279→        ) : (
   280→          <motion.section variants={stagger} initial="hidden" animate="visible" className="px-4 mt-4">
   281→            <motion.div variants={fadeUp} className="rounded-xl border border-emerald-200/50 bg-white/80 shadow-sm backdrop-blur-sm dark:border-emerald-800/50 dark:bg-card/80 p-4">
   282→              <p className="text-xs text-muted-foreground mb-1">{t('finance.overview')}</p>
   283→              <p className={`text-2xl font-bold ${balance >= 0 ? 'text-emerald-600 dark:text-emerald-400' : 'text-rose-600 dark:text-rose-400'}`}>{formatRupiah(balance)}</p>
   284→            </motion.div>
   285→            <div className="grid grid-cols-2 gap-3 mt-3">
   286→              {[
   287→                { label: t('finance.income'), val: monthlyIn, icon: ArrowUpRight, bg: 'bg-emerald-50 dark:bg-emerald-500/10', cls: 'text-emerald-600 dark:text-emerald-400' },
   288→                { label: t('finance.expense'), val: monthlyOut, icon: ArrowDownRight, bg: 'bg-rose-50 dark:bg-rose-500/10', cls: 'text-rose-600 dark:text-rose-400' },
   289→              ].map((s) => (
   290→                <motion.div key={s.label} variants={fadeUp} className="rounded-xl border border-emerald-200/50 bg-white/80 shadow-sm backdrop-blur-sm dark:border-emerald-800/50 dark:bg-card/80 p-3 flex items-center gap-3">
   291→                  <div className={`flex h-9 w-9 shrink-0 items-center justify-center rounded-lg ${s.bg}`}><s.icon className={`h-4 w-4 ${s.cls}`} /></div>
   292→                  <div><p className="text-[10px] text-muted-foreground uppercase tracking-wider">{s.label}</p><p className={`text-sm font-bold ${s.cls}`}>{formatRupiah(s.val)}</p></div>
   293→                </motion.div>
   294→              ))}
   295→            </div>
   296→          </motion.section>
   297→        )}
   298→
   299→        {/* ── Tab Bar ────────────────────────────────── */}
   300→        <div className="mx-4 mt-4">
   301→          <div className="flex rounded-xl border border-emerald-200/50 bg-white/80 shadow-sm backdrop-blur-sm dark:border-emerald-800/50 dark:bg-card/80 p-1">
   302→            {(['records', 'budgets', 'goals', 'report', 'recurring'] as const).map((tab) => (
   303→              <button key={tab} onClick={() => setActiveTab(tab as any)}
   304→                className={`flex-1 rounded-lg py-2 text-[10px] sm:text-xs font-semibold transition-all ${activeTab === tab ? 'nm-raised-sm bg-emerald-600 text-white' : 'text-muted-foreground hover:text-foreground'}`}>
   305→                {tab === 'records' ? t('finance.records') : tab === 'budgets' ? t('finance.budgets') : tab === 'goals' ? t('finance.goals') : tab === 'report' ? 'Laporan' : 'Berulang'}
   306→              </button>
   307→            ))}
   308→          </div>
   309→        </div>
   310→
   311→        {/* ── Tab Content ───────────────────────────── */}
   312→        <AnimatePresence mode="wait">
   313→          <motion.div key={activeTab} initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -8 }} transition={{ duration: 0.25 }} className="px-4 mt-4">
   314→            {activeTab === 'records' && <RecordsSection grouped={grouped} loading={loading} onAdd={() => { resetRec(); setDlgRecord(true); }} onEdit={openEditRec} onDelete={(id) => setDlgDelete({ type: 'record', id })} />}
   315→            {activeTab === 'budgets' && <BudgetsSection budgets={budgetsWithSpent} loading={loading} onAdd={() => { resetBud(); setDlgBudget(true); }} onEdit={openEditBud} onDelete={(id) => setDlgDelete({ type: 'budget', id })} />}
   316→            {activeTab === 'goals' && <GoalsSection goals={goals} loading={loading} onAdd={() => { resetGo(); setDlgGoal(true); }} onEdit={openEditGo} onDelete={(id) => setDlgDelete({ type: 'goal', id })} onAddSavings={(g) => { setDlgSavings(g); setSavAmt(''); }} />}
   317→            {activeTab === 'report' && userId && <FinanceReportTab userId={userId} />}
   318→            {activeTab === 'recurring' && userId && <FinanceRecurringTab userId={userId} />}
   319→          </motion.div>
   320→        </AnimatePresence>
   321→      </div>
   322→
   323→      {/* ── Dialogs ──────────────────────────────────── */}
   324→      <RecordFormDialog open={dlgRecord} onClose={() => { setDlgRecord(false); resetRec(); }} recType={recType} setRecType={setRecType} recAmt={recAmt} setRecAmt={setRecAmt} recCat={recCat} setRecCat={setRecCat} recDesc={recDesc} setRecDesc={setRecDesc} recDate={recDate} setRecDate={setRecDate} isEdit={!!editRecId} onSave={saveRec} />
   325→      <BudgetFormDialog open={dlgBudget} onClose={() => { setDlgBudget(false); resetBud(); }} cat={budCat} setCat={setBudCat} limit={budLimit} setLimit={setBudLimit} period={budPeriod} setPeriod={setBudPeriod} isEdit={!!editBudId} onSave={saveBud} />
   326→      <GoalFormDialog open={dlgGoal} onClose={() => { setDlgGoal(false); resetGo(); }} title={goTitle} setTitle={setGoTitle} target={goTarget} setTarget={setGoTarget} saved={goSaved} setSaved={setGoSaved} deadline={goDeadline} setDeadline={setGoDeadline} icon={goIcon} setIcon={setGoIcon} isEdit={!!editGoId} onSave={saveGoal} />
   327→
   328→      <Dialog open={!!dlgSavings} onOpenChange={() => { setDlgSavings(null); setSavAmt(''); }}>
   329→        <DialogContent className="rounded-2xl sm:max-w-sm">
   330→          <DialogHeader>
   331→            <DialogTitle className="flex items-center gap-2"><PiggyBank className="h-5 w-5 text-emerald-500" />{t('finance.addSavings')}</DialogTitle>
   332→            <DialogDescription>{dlgSavings?.title}</DialogDescription>
   333→          </DialogHeader>
