    80→/* ── Swipe-to-delete User Card ──────────────────────────────── */
    81→
    82→interface SwipeableUserCardProps {
    83→  user: AdminUser;
    84→  onEdit: (user: AdminUser) => void;
    85→  onDelete: (user: AdminUser) => void;
    86→}
    87→
    88→function SwipeableUserCard({ user, onEdit, onDelete }: SwipeableUserCardProps) {
    89→  const x = useMotionValue(0);
    90→  const deleteOpacity = useTransform(x, [-80, -40], [1, 0]);
    91→  const deleteScale = useTransform(x, [-80, -40], [1, 0.8]);
    92→
    93→  const handleDragEnd = (
    94→    _event: MouseEvent | TouchEvent | PointerEvent,
    95→    info: PanInfo
    96→  ) => {
    97→    if (info.offset.x < -60) {
    98→      onDelete(user);
    99→    } else {
   100→      x.set(0);
   101→    }
   102→  };
   103→
   104→  const initials = (user.name || user.email)
   105→    .split(' ')
   106→    .map((w) => w[0])
   107→    .join('')
   108→    .toUpperCase()
   109→    .slice(0, 2);
   110→
   111→  return (
   112→    <div className="relative overflow-hidden rounded-2xl mb-3">
   113→      {/* Red delete background */}
   114→      <motion.div
   115→        className="absolute inset-0 z-0 flex items-center justify-end pr-5 bg-gradient-to-r from-transparent to-red-500 rounded-2xl"
   116→        style={{ opacity: deleteOpacity }}
   117→      >
   118→        <motion.div
   119→          style={{ scale: deleteScale }}
   120→          className="flex items-center gap-2 text-white font-semibold text-sm"
   121→        >
   122→          <Trash2 className="h-5 w-5" />
   123→          Hapus
   124→        </motion.div>
   125→      </motion.div>
   126→
   127→      {/* Card content */}
   128→      <motion.div
   129→        className="relative z-10 cursor-grab active:cursor-grabbing"
   130→        style={{ x }}
   131→        drag="x"
   132→        dragConstraints={{ left: -80, right: 0 }}
   133→        dragElastic={{ left: 0.1, right: 0 }}
   134→        onDragEnd={handleDragEnd}
   135→      >
   136→        <Card className="border-border/50 bg-card/90 backdrop-blur-sm transition-shadow hover:shadow-md">
   137→          <CardContent className="p-4">
   138→            <div className="flex items-center gap-3">
   139→              {/* Avatar */}
   140→              <div className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl bg-gradient-to-br from-emerald-500 to-teal-600 text-white text-sm font-bold">
   141→                {initials}
   142→              </div>
   143→
   144→              {/* Info */}
   145→              <div className="flex-1 min-w-0">
   146→                <div className="flex items-center gap-2">
   147→                  <h3 className="text-sm font-semibold truncate">
   148→                    {user.name || 'Tanpa Nama'}
   149→                  </h3>
   150→                  {/* Status dot */}
   151→                  <span className="relative flex h-2.5 w-2.5 shrink-0">
   152→                    {user.isActive ? (
   153→                      <>
   154→                        <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-emerald-400 opacity-75" />
   155→                        <span className="relative inline-flex rounded-full h-2.5 w-2.5 bg-emerald-500" />
   156→                      </>
   157→                    ) : (
   158→                      <span className="relative inline-flex rounded-full h-2.5 w-2.5 bg-gray-400" />
   159→                    )}
   160→                  </span>
   161→                </div>
   162→                <p className="mt-0.5 text-xs text-muted-foreground truncate">
   163→                  {user.email}
   164→                </p>
   165→                <div className="mt-1 flex items-center gap-1.5">
   166→                  <Badge
   167→                    variant="secondary"
   168→                    className={`text-[10px] px-1.5 ${
   169→                      user.role === 'admin'
   170→                        ? 'bg-purple-500/15 text-purple-600 dark:text-purple-400 border-purple-200 dark:border-purple-800/40'
   171→                        : 'bg-muted text-muted-foreground'
   172→                    }`}
   173→                  >
   174→                    {user.role === 'admin' ? (
   175→                      <span className="flex items-center gap-0.5">
   176→                        <ShieldCheck className="h-3 w-3" />
   177→                        Admin
   178→                      </span>
   179→                    ) : (
   180→                      'User'
   181→                    )}
   182→                  </Badge>
   183→                  <span className="text-[10px] text-muted-foreground">
   184→                    {new Date(user.createdAt).toLocaleDateString('id-ID', {
   185→                      day: 'numeric',
   186→                      month: 'short',
   187→                      year: 'numeric',
   188→                    })}
   189→                  </span>
   190→                </div>
   191→              </div>
   192→
   193→              {/* Edit button */}
   194→              <motion.button
   195→                whileTap={{ scale: 0.9 }}
   196→                onClick={(e) => {
   197→                  e.stopPropagation();
   198→                  onEdit(user);
   199→                }}
   200→                className="flex h-9 w-9 shrink-0 items-center justify-center rounded-xl border border-border/40 bg-muted/50 transition-colors hover:bg-accent"
   201→                aria-label="Edit user"
   202→              >
   203→                <Edit3 className="h-4 w-4 text-muted-foreground" />
   204→              </motion.button>
   205→            </div>
   206→          </CardContent>
   207→        </Card>
   208→      </motion.div>
   209→    </div>
   210→  );
   211→}

   284→          (u.name || '').toLowerCase().includes(q) ||
   285→          u.email.toLowerCase().includes(q)
   286→      );
   287→    }
   288→
   289→    return result;

   291→

   311→          'Content-Type': 'application/json',
   312→          'X-Admin-Key': 'dapurmind2025',

   315→          id: editingUser.id,
   316→          name: editName.trim(),
   317→          email: editEmail.trim(),
   318→          role: editRole,
   319→          isActive: editIsActive,
   320→        }),
   321→      });

   323→        const data = await res.json();

   325→      }
   326→      setShowEditDialog(false);
   327→      setEditingUser(null);
   328→      setRefreshKey((k) => k + 1);

   330→      setError(err instanceof Error ? err.message : 'Gagal mengupdate user');

   396→        <motion.div variants={fadeUp} className="mb-5">
   397→          <div className="flex items-center gap-3">
   398→            <motion.button
