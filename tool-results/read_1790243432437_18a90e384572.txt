    88→/* ── Difficulty color helper ─────────────────────────────────── */
    89→
    90→function getDifficultyColor(difficulty: string) {
    91→  switch (difficulty) {
    92→    case 'Mudah':
    93→      return 'bg-emerald-500/15 text-emerald-600 dark:text-emerald-400 border-emerald-200 dark:border-emerald-800/40';
    94→    case 'Sedang':
    95→      return 'bg-amber-500/15 text-amber-600 dark:text-amber-400 border-amber-200 dark:border-amber-800/40';
    96→    case 'Susah':
    97→      return 'bg-red-500/15 text-red-600 dark:text-red-400 border-red-200 dark:border-red-800/40';
    98→    default:
    99→      return '';
   100→  }
   101→}

   103→/* ── Swipe-to-delete Recipe Card ────────────────────────────── */
   104→
   105→interface SwipeableRecipeCardProps {
   106→  recipe: AdminRecipe;
   107→  onEdit: (recipe: AdminRecipe) => void;
   108→  onDelete: (recipe: AdminRecipe) => void;
   109→}
   110→
   111→function SwipeableRecipeCard({
   112→  recipe,
   113→  onEdit,
   114→  onDelete,
   115→}: SwipeableRecipeCardProps) {
   116→  const x = useMotionValue(0);
   117→  const deleteOpacity = useTransform(x, [-80, -40], [1, 0]);
   118→  const deleteScale = useTransform(x, [-80, -40], [1, 0.8]);
   119→
   120→  const handleDragEnd = (
   121→    _event: MouseEvent | TouchEvent | PointerEvent,
   122→    info: PanInfo
   123→  ) => {
   124→    if (info.offset.x < -60) {
   125→      onDelete(recipe);
   126→    } else {
   127→      x.set(0);
   128→    }
   129→  };
   130→
   131→  const totalTime = recipe.cookTime + recipe.prepTime;
   132→
   133→  return (
   134→    <div className="relative overflow-hidden rounded-2xl mb-3">
   135→      {/* Red delete background */}
   136→      <motion.div
   137→        className="absolute inset-0 z-0 flex items-center justify-end pr-5 bg-gradient-to-r from-transparent to-red-500 rounded-2xl"
   138→        style={{ opacity: deleteOpacity }}
   139→      >
   140→        <motion.div
   141→          style={{ scale: deleteScale }}
   142→          className="flex items-center gap-2 text-white font-semibold text-sm"
   143→        >
   144→          <Trash2 className="h-5 w-5" />
   145→          Hapus
   146→        </motion.div>
   147→      </motion.div>
   148→
   149→      {/* Card content */}
   150→      <motion.div
   151→        className="relative z-10 cursor-grab active:cursor-grabbing"
   152→        style={{ x }}
   153→        drag="x"
   154→        dragConstraints={{ left: -80, right: 0 }}
   155→        dragElastic={{ left: 0.1, right: 0 }}
   156→        onDragEnd={handleDragEnd}
   157→      >
   158→        <Card className="border-border/50 bg-card/90 backdrop-blur-sm transition-shadow hover:shadow-md">
   159→          <CardContent className="p-4">
   160→            <div className="flex items-start gap-3">
   161→              {/* Icon */}
   162→              <div className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl bg-amber-500/15">
   163→                <ChefHat className="h-5 w-5 text-amber-600" />
   164→              </div>
   165→
   166→              {/* Info */}
   167→              <div className="flex-1 min-w-0">
   168→                <div className="flex items-center gap-2">
   169→                  <h3 className="text-sm font-semibold truncate">
   170→                    {recipe.name}
   171→                  </h3>
   172→                  {recipe.isPublished ? (
   173→                    <Eye className="h-3.5 w-3.5 text-emerald-500 shrink-0" />
   174→                  ) : (
   175→                    <EyeOff className="h-3.5 w-3.5 text-muted-foreground shrink-0" />
   176→                  )}
   177→                </div>
   178→                <p className="mt-0.5 text-xs text-muted-foreground truncate">
   179→                  oleh {recipe.user?.name || recipe.user?.email || 'Anonim'}
   180→                </p>
   181→                <div className="mt-1.5 flex flex-wrap items-center gap-1.5">
   182→                  <Badge
   183→                    variant="secondary"
   184→                    className={`text-[10px] px-1.5 border ${getDifficultyColor(
   185→                      recipe.difficulty
   186→                    )}`}
   187→                  >
   188→                    <Flame className="h-3 w-3 mr-0.5" />
   189→                    {recipe.difficulty}
   190→                  </Badge>
   191→                  <span className="text-[10px] text-muted-foreground flex items-center gap-0.5">
   192→                    <Clock className="h-3 w-3" />
   193→                    {totalTime} menit
   194→                  </span>
   195→                  <span className="text-[10px] text-muted-foreground flex items-center gap-0.5">
   196→                    <Heart className="h-3 w-3" />
   197→                    {recipe.likes}
   198→                  </span>
   199→                </div>
   200→                <div className="mt-1 flex items-center gap-2">
   201→                  <Badge variant="secondary" className="text-[10px] px-1.5">
   202→                    {recipe.category}
   203→                  </Badge>
   204→                  <span className="text-[10px] text-muted-foreground">
   205→                    {new Date(recipe.createdAt).toLocaleDateString('id-ID', {
   206→                      day: 'numeric',
   207→                      month: 'short',
   208→                      year: 'numeric',
   209→                    })}
   210→                  </span>
   211→                </div>
   212→              </div>
   213→
   214→              {/* Edit button */}
   215→              <motion.button
   216→                whileTap={{ scale: 0.9 }}
   217→                onClick={(e) => {
   218→                  e.stopPropagation();
   219→                  onEdit(recipe);
   220→                }}
   221→                className="flex h-9 w-9 shrink-0 items-center justify-center rounded-xl border border-border/40 bg-muted/50 transition-colors hover:bg-accent"
   222→                aria-label="Edit resep"
   223→              >
   224→                <Edit3 className="h-4 w-4 text-muted-foreground" />
   225→              </motion.button>
   226→            </div>
   227→          </CardContent>
   228→        </Card>
   229→      </motion.div>
   230→    </div>
   231→  );
   232→}

   234→/* ── Main Component ──────────────────────────────────────────── */
   235→
   236→export function AdminRecipeManager() {
   237→  const setScreen = useAppStore((s) => s.setScreen);
   238→  const goBack = useAppStore((s) => s.goBack);
   239→
   240→  /* ── Local state ───────────────────────────────────────────── */
   241→  const [loading, setLoading] = useState(true);
   242→  const [error, setError] = useState<string | null>(null);
   243→  const [recipes, setRecipes] = useState<AdminRecipe[]>([]);
   244→  const [search, setSearch] = useState('');
   245→  const [categoryFilter, setCategoryFilter] = useState('all');
   246→  const [saving, setSaving] = useState(false);
   247→  const [deleting, setDeleting] = useState(false);
   248→  const [refreshKey, setRefreshKey] = useState(0);
   249→
   250→  // Edit dialog state
   251→  const [showEditDialog, setShowEditDialog] = useState(false);
   252→  const [editingRecipe, setEditingRecipe] = useState<AdminRecipe | null>(null);
   253→  const [editName, setEditName] = useState('');
   254→  const [editDescription, setEditDescription] = useState('');
   255→  const [editCategory, setEditCategory] = useState('');
   256→  const [editDifficulty, setEditDifficulty] = useState('Mudah');
   257→  const [editCookTime, setEditCookTime] = useState(30);
   258→  const [editPrepTime, setEditPrepTime] = useState(15);
   259→  const [editIsPublished, setEditIsPublished] = useState(false);
   260→
   261→  // Delete dialog state
   262→  const [showDeleteDialog, setShowDeleteDialog] = useState(false);
   263→  const [deletingRecipe, setDeletingRecipe] = useState<AdminRecipe | null>(null);
   264→
   265→  /* ── Fetch recipes ──────────────────────────────────────────── */
   266→  const fetchRecipes = useCallback(async () => {
   267→    try {
   268→      setLoading(true);
   269→      setError(null);
   270→      const res = await fetch('/api/admin/recipes', {
   271→        headers: { 'X-Admin-Key': 'dapurmind2025' },
   272→      });
   273→      if (!res.ok) throw new Error('Gagal memuat data resep');
   274→      const data = await res.json();
   275→      setRecipes(data.recipes || []);
   276→    } catch (err) {
   277→      setError(err instanceof Error ? err.message : 'Terjadi kesalahan');
   278→    } finally {
   279→      setLoading(false);
   280→    }
   281→  }, []);
   282→
   283→  useEffect(() => {
   284→    fetchRecipes();
   285→  }, [fetchRecipes, refreshKey]);
   286→
   287→  /* ── Stats ─────────────────────────────────────────────────── */
   288→  const stats = useMemo(() => {
   289→    const nonDeleted = recipes.filter((r) => !r.deletedAt);
   290→    return {
   291→      total: nonDeleted.length,
   292→      published: nonDeleted.filter((r) => r.isPublished).length,
   293→      draft: nonDeleted.filter((r) => !r.isPublished).length,
   294→      deleted: recipes.filter((r) => r.deletedAt).length,
   295→    };
   296→  }, [recipes]);
   297→
   298→  /* ── Categories ────────────────────────────────────────────── */
   299→  const categories = useMemo(() => {
   300→    const cats = new Set(recipes.map((r) => r.category));
   301→    return ['all', ...Array.from(cats).sort()];
   302→  }, [recipes]);
   303→
   304→  /* ── Filtered recipes ─────────────────────────────────────── */
   305→  const filteredRecipes = useMemo(() => {
   306→    let result = recipes.filter((r) => !r.deletedAt);
   307→
   308→    if (categoryFilter !== 'all') {
   309→      result = result.filter((r) => r.category === categoryFilter);
   310→    }
   311→
   312→    if (search.trim()) {
   313→      const q = search.toLowerCase();
   314→      result = result.filter(
   315→        (r) =>
   316→          r.name.toLowerCase().includes(q) ||
   317→          (r.user?.name || '').toLowerCase().includes(q) ||
   318→          r.category.toLowerCase().includes(q)
   319→      );
   320→    }
   321→
   322→    return result;
   323→  }, [recipes, categoryFilter, search]);
   324→
   325→  /* ── Dialog handlers ───────────────────────────────────────── */
   326→
   327→  const openEditDialog = useCallback((recipe: AdminRecipe) => {
   328→    setEditingRecipe(recipe);
   329→    setEditName(recipe.name);
   330→    setEditDescription(recipe.description);
   331→    setEditCategory(recipe.category);
   332→    setEditDifficulty(recipe.difficulty);
   333→    setEditCookTime(recipe.cookTime);
   334→    setEditPrepTime(recipe.prepTime);
   335→    setEditIsPublished(recipe.isPublished);
   336→    setShowEditDialog(true);
   337→  }, []);
   338→
   339→  const handleEditSubmit = useCallback(async () => {
   340→    if (!editingRecipe) return;
   341→    try {
   342→      setSaving(true);
   343→      setError(null);
   344→      const res = await fetch('/api/admin/recipes', {
   345→        method: 'PUT',
   346→        headers: {
   347→          'Content-Type': 'application/json',
   348→          'X-Admin-Key': 'dapurmind2025',
   349→        },
   350→        body: JSON.stringify({
   351→          id: editingRecipe.id,
   352→          name: editName.trim(),
   353→          description: editDescription.trim(),
   354→          category: editCategory,
   355→          difficulty: editDifficulty,
   356→          cookTime: editCookTime,
   357→          prepTime: editPrepTime,
   358→          isPublished: editIsPublished,
   359→        }),
   360→      });
   361→      if (!res.ok) {
   362→        const data = await res.json();
   363→        throw new Error(data.error || 'Gagal mengupdate resep');
   364→      }
   365→      setShowEditDialog(false);
   366→      setEditingRecipe(null);
   367→      setRefreshKey((k) => k + 1);
   368→    } catch (err) {
   369→      setError(err instanceof Error ? err.message : 'Gagal mengupdate resep');
   370→    } finally {
   371→      setSaving(false);
   372→    }
   373→  }, [
   374→    editingRecipe,
   375→    editName,
   376→    editDescription,
   377→    editCategory,
   378→    editDifficulty,
   379→    editCookTime,
   380→    editPrepTime,
   381→    editIsPublished,
   382→  ]);
   383→
   384→  const openDeleteDialog = useCallback((recipe: AdminRecipe) => {
   385→    setDeletingRecipe(recipe);
   386→    setShowDeleteDialog(true);
   387→  }, []);
   388→
   389→  const handleDeleteConfirm = useCallback(async () => {
   390→    if (!deletingRecipe) return;
   391→    try {
   392→      setDeleting(true);
   393→      setError(null);
   394→      const res = await fetch('/api/admin/recipes', {
   395→        method: 'DELETE',
   396→        headers: {
   397→          'Content-Type': 'application/json',
   398→          'X-Admin-Key': 'dapurmind2025',
   399→        },
   400→        body: JSON.stringify({ id: deletingRecipe.id }),
   401→      });
   402→      if (!res.ok) {
   403→        const data = await res.json();
   404→        throw new Error(data.error || 'Gagal menghapus resep');
   405→      }
   406→      setShowDeleteDialog(false);
   407→      setDeletingRecipe(null);
   408→      setRefreshKey((k) => k + 1);
   409→    } catch (err) {
   410→      setError(err instanceof Error ? err.message : 'Gagal menghapus resep');
   411→    } finally {
   412→      setDeleting(false);
   413→    }
   414→  }, [deletingRecipe]);
   415→
   416→  /* ── Header nav items ─────────────────────────────────────── */
   417→  const headerNav = [
   418→    { icon: LayoutDashboard, screen: 'admin-dashboard' as const, label: 'Dashboard' },
   419→    { icon: Users, screen: 'admin-users' as const, label: 'User' },
   420→    { icon: Trash2, screen: 'admin-trash' as const, label: 'Sampah' },
   421→    { icon: Link2, screen: 'admin-affiliate' as const, label: 'Afiliasi' },
   422→    { icon: Megaphone, screen: 'admin-ads' as const, label: 'Iklan' },
   423→    { icon: BarChart3, screen: 'admin-analytics' as const, label: 'Analitik' },
   424→  ];
   425→
   426→  /* ── Render ────────────────────────────────────────────────── */
   427→
   428→  return (
   429→    <div className="relative min-h-screen pb-28 bg-gradient-to-br from-emerald-50/30 via-white to-amber-50/20 dark:from-emerald-950/20 dark:via-background dark:to-amber-950/10">
   430→      <motion.div
   431→        variants={stagger}
   432→        initial="hidden"
   433→        animate="visible"
   434→        className="relative z-10 mx-auto max-w-lg px-4 pt-4"
   435→      >
   436→        {/* ── Header ──────────────────────────────────────────── */}
   437→        <motion.div variants={fadeUp} className="mb-5">
   438→          <div className="flex items-center gap-3">
   439→            <motion.button
   440→              whileTap={{ scale: 0.9 }}
   441→              onClick={goBack}
   442→              className="flex h-10 w-10 items-center justify-center rounded-xl nm-raised transition-colors hover:bg-accent"
   443→              aria-label="Kembali"
   444→            >
   445→              <ArrowLeft className="h-5 w-5" />
   446→            </motion.button>
   447→            <div className="flex-1 min-w-0">
   448→              <h1 className="text-lg font-bold tracking-tight text-emerald-600 dark:text-emerald-400 leading-tight">
   449→                Manajemen Resep
   450→              </h1>
   451→              <p className="mt-0.5 text-xs text-muted-foreground">
   452→                Kelola semua resep kreator
   453→              </p>
   454→            </div>
   455→            <div className="flex items-center gap-1.5">
   456→              {headerNav.map((item) => (
   457→                <motion.button
   458→                  key={item.screen}
   459→                  whileTap={{ scale: 0.9 }}
   460→                  onClick={() => setScreen(item.screen)}
   461→                  className="flex h-9 w-9 items-center justify-center rounded-xl nm-raised transition-colors hover:bg-accent"
   462→                  aria-label={item.label}
   463→                >
   464→                  <item.icon className="h-4 w-4 text-muted-foreground" />
   465→                </motion.button>

   467→              <motion.button
   468→                whileTap={{ scale: 0.9 }}
   469→                onClick={() => {
   470→                  useAppStore.getState().setAdminLoggedIn(false);
   471→                  useAppStore.getState().setScreen('dashboard');
   472→                }}
   473→                className="flex h-9 w-9 items-center justify-center rounded-xl border border-red-200 bg-red-50 shadow-sm transition-colors hover:bg-red-100 dark:border-red-500/20 dark:bg-red-500/10 dark:hover:bg-red-500/20"

   482→        {/* ── Error Banner ────────────────────────────────────── */}
   483→        <AnimatePresence>

   487→              animate={{ opacity: 1, y: 0 }}
   488→              exit={{ opacity: 0, y: -10 }}
   489→              className="mb-4 rounded-xl border border-red-200 bg-red-50 p-3 dark:border-red-800/40 dark:bg-red-500/10"
   490→            >
   491→              <div className="flex items-start gap-2">
   492→                <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0 text-red-500" />
   493→                <div className="flex-1 min-w-0">
   494→                  <p className="text-xs font-medium text-red-700 dark:text-red-400">
   495→                    {error}
   496→                  </p>
   497→                </div>
   498→                <button
   499→                  onClick={() => setError(null)}
   500→                  className="shrink-0 text-red-400 hover:text-red-600"
   501→                >
   502→                  <XCircle className="h-4 w-4" />
   503→                </button>
   504→              </div>
   505→            </motion.div>
   506→          )}
   507→        </AnimatePresence>
   508→
   509→        {/* ── Stats Banner ───────────────────────────────────── */}
   510→        <motion.div variants={fadeUp} className="mb-5">
   511→          <div className="rounded-2xl border border-border/40 bg-card/80 backdrop-blur-sm p-4">
   512→            <div className="grid grid-cols-4 gap-2">
   513→              <div className="text-center">
   514→                <span className="text-base font-bold text-emerald-600 dark:text-emerald-400">
   515→                  {stats.total}
   516→                </span>
   517→                <p className="mt-0.5 text-[10px] text-muted-foreground">Total</p>
   518→              </div>
   519→              <div className="text-center">

   521→                <p className="mt-0.5 text-[10px] text-muted-foreground">Terbit</p>
   522→              </div>
   523→              <div className="text-center">
   524→                <span className="text-base font-bold text-amber-600">
   525→                  {stats.draft}
   526→                </span>
   527→                <p className="mt-0.5 text-[10px] text-muted-foreground">Draft</p>
   528→              </div>
   529→              <div className="text-center">
   530→                <span className="text-base font-bold text-red-500">
   531→                  {stats.deleted}
   532→                </span>
   533→                <p className="mt-0.5 text-[10px] text-muted-foreground">Dihapus</p>
   534→              </div>
   535→            </div>
   536→          </div>
   537→        </motion.div>
   538→
   539→        {/* ── Search Bar ──────────────────────────────────────── */}
   540→        <motion.div variants={fadeUp} className="mb-4">
   541→          <div className="relative">
   542→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground/50" />
   543→            <Input
   544→              placeholder="Cari resep berdasarkan nama, penulis, kategori..."
   545→              value={search}
   546→              onChange={(e) => setSearch(e.target.value)}
   547→              className="rounded-xl pl-10"
   548→            />
   549→          </div>
   550→        </motion.div>
   551→
   552→        {/* ── Category Filter ──────────────────────────────────── */}
   553→        <motion.div variants={fadeUp} className="mb-4">

   629→                  <span className="text-3xl">🍳</span>
   630→                </div>
   631→                <h3 className="text-base font-semibold">
   632→                  {search.trim() || categoryFilter !== 'all'
   633→                    ? 'Resep Tidak Ditemukan'

   780→                  type="number"
